// ── Domain Types ─────────────────────────────────────────────────────
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TelemetryEvent {
    pub robot_id: u32,
    pub timestamp_ns: u64,
    pub temp: f64,
    pub z_velocity: f64,
    /// 0=Nominal 1=Overheating 2=Halted 3=IpcOverflow
    pub status: u8,
    pub ack_command_id: Option<[u8; 16]>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CommandEvent {
    pub robot_id: u32,
    pub command_id: [u8; 16], // UUIDv4
    pub sequence: u64,        // Per-robot monotonic counter — replay protection
    /// 1=CoolDown 2=EmergencyHalt
    pub action: u8,
    pub duration_sec: u32,
    pub issued_at_ns: u64,
    pub valid_until_ns: u64, // Absolute deadline
    pub min_epoch: u64,
    pub max_epoch: u64,
}

/// Published to TWIN_STATE_TOPIC for Grafana / external monitoring, and
/// (V3.1) replayed on worker startup to warm-restore `RobotTwinState`
/// after a restart — see `qib_cloud_backend::warmup`.
///
/// `variance` and `next_sequence` were added for that replay purpose:
/// `temp_stddev` alone (a derived, sqrt'd display value) isn't enough to
/// reconstruct the exact EWMA accumulator, and `next_sequence` is what
/// prevents a restarted backend from re-issuing command sequence numbers
/// the microkernel's replay-protection has already seen and would reject.
///
/// Wire-format note: these two fields are appended at the end of the
/// struct, so postcard's positional encoding means older (pre-V3.1)
/// snapshots simply fail to decode as this newer type (a clean `Err`, not
/// silent corruption) rather than partially/incorrectly parsing — callers
/// already treat individual snapshot decode failures as "skip and log",
/// which is the correct behavior for stale-format history during replay.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TwinStateSnapshot {
    pub robot_id: u32,
    pub temp_mean: f64,
    pub temp_stddev: f64,
    pub z_score: f64,
    pub breaker_open: bool,
    pub timestamp_ns: u64,
    /// Raw EWMA variance accumulator (Welford's `m2`), not the sqrt'd
    /// `temp_stddev` above — needed to restore the twin exactly.
    pub variance: f64,
    /// Next command sequence number this robot's twin will issue.
    pub next_sequence: u64,
}

// ── V3-1: Wire-format helpers (postcard) ─────────────────────────────

/// Serialize any message to a compact binary vec using postcard.
/// Used for all Kafka payloads and IPC buffers.
pub fn serialize_msg<T: Serialize>(msg: &T) -> Vec<u8> {
    postcard::to_allocvec(msg).unwrap_or_default()
}

/// Deserialize a postcard-encoded message.
pub fn deserialize_msg<'a, T: Deserialize<'a>>(data: &'a [u8]) -> Result<T, postcard::Error> {
    postcard::from_bytes(data)
}
