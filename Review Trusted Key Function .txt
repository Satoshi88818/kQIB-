//! Edge proxy: bridges async Kafka I/O with the bare-metal microkernel via
//! lock-free shared memory. Performs Ed25519 verification, TTL checks, epoch
//! validation, and per-robot rate limiting before touching the IPC ring.
//!
//! V3 additions:
//! - ClockSync: EWMA drift correction keeps TTL checks accurate
//!   despite NTP-class skew between cloud and robot.
//! - yield_now(): Event-driven telemetry loop (was 10ms sleep).
//! - postcard: Binary wire format (via qib_schema helpers).
//! - Contextual sig: Verify hash(robot_id ‖ command_id ‖ action ‖ sequence).
//!
//! V3.1 fixes:
//! - load_trusted_keys() now actually parses TRUSTED_PUBKEYS instead of
//!   returning an empty Vec (which silently disabled verification).
//! - The IPC ring push/poll that were previously commented-out stubs are
//!   now wired to a real POSIX-shared-memory-backed SharedRingBuffer (see
//!   `shm.rs`), using qib_muk's ring buffer directly.

const ROBOT_ID: u32 = 42;

// ── V3-3: Clock Synchronisation ────────────────────────────────────────
//
// Cloud timestamps and robot-local monotonic clock may diverge by tens of
// milliseconds. A naïve `now_ns > valid_until_ns` check will incorrectly
// expire commands when the robot clock is ahead of the cloud.
//
// ClockSync maintains an EWMA of the observed drift so that TTL checks use
// an adjusted "cloud-equivalent now".
struct ClockSync {
    /// Smoothed offset: cloud_time - robot_time (signed, nanoseconds).
    drift_offset_ns: i64,
}

/// "Smaller fixes" round: bounds how much any single observation can move
/// `drift_offset_ns`. A corrupted or malicious `issued_at_ns` (e.g. near
/// `u64::MAX`) previously fed straight into `drift_offset_ns * 9`
/// unchecked — implausible input could push that multiply toward i64
/// overflow. No real clock (even a badly unsynced one) should be off by
/// more than about a minute; anything beyond that is far more likely a
/// corrupted timestamp than genuine drift, so it's clamped and logged
/// rather than trusted.
const MAX_PLAUSIBLE_DRIFT_NS: i64 = 60 * 1_000_000_000; // ±60 seconds

impl ClockSync {
    fn new() -> Self {
        Self { drift_offset_ns: 0 }
    }

    /// Call once per received command with the cloud-issued timestamp and the
    /// robot's local receive time to update the EWMA.
    fn synchronize(&mut self, cloud_issued_ns: u64, proxy_received_ns: u64) {
        // Subtract in i128 first: a pathological `cloud_issued_ns` (e.g.
        // near u64::MAX from a corrupted or malicious command) could
        // otherwise wrap unpredictably when cast straight to i64 before
        // we've had a chance to bound it.
        let raw_drift_i128 = cloud_issued_ns as i128 - proxy_received_ns as i128;
        let bound = MAX_PLAUSIBLE_DRIFT_NS as i128;
        let bounded = raw_drift_i128.clamp(-bound, bound);

        if bounded != raw_drift_i128 {
            warn!(
                "Implausible clock drift observed ({raw_drift_i128}ns, outside \
                 ±{MAX_PLAUSIBLE_DRIFT_NS}ns) — clamped to {bounded}ns. This may \
                 indicate a corrupted command timestamp; investigate if it recurs."
            );
        }
        let observed_drift = bounded as i64;

        // EWMA α = 0.1 (smooth out network jitter). `drift_offset_ns` stays
        // bounded within ±MAX_PLAUSIBLE_DRIFT_NS by induction — a weighted
        // average of two already-bounded values can't leave that range —
        // so this multiply can't overflow i64 in practice; `saturating_mul`
        // is a defensive backstop, not load-bearing, in case that
        // invariant is ever violated by a future change.
        self.drift_offset_ns = (self.drift_offset_ns.saturating_mul(9) + observed_drift) / 10;
    }

    /// Returns the robot's local clock adjusted to approximate cloud time.
    fn adjusted_now_ns(&self) -> u64 {
        (raw_time_ns() as i64 + self.drift_offset_ns) as u64
    }
}

/// Parses trusted Ed25519 public keys from a raw string: comma-separated
/// 32-byte hex strings (e.g. from the `TRUSTED_PUBKEYS` env var). Malformed
/// entries are logged and skipped rather than aborting the whole list, so a
/// single typo'd key doesn't take down verification for every other key.
///
/// Pure and side-effect-free (aside from logging) so it's directly unit
/// testable without touching the environment.
fn parse_trusted_keys(raw: &str) -> Vec<VerifyingKey> {
    raw.split(',')
        .map(str::trim)
        .filter(|s| !s.is_empty())
        .filter_map(|s| match hex::decode(s) {
            Ok(bytes) => match <[u8; 32]>::try_from(bytes.as_slice()) {
                Ok(arr) => match VerifyingKey::from_bytes(&arr) {
                    Ok(key) => Some(key),
                    Err(e) => {
                        error!("Invalid trusted pubkey '{}': {}", s, e);
                        None
                    }
                },
                Err(_) => {
                    error!(
                        "Trusted pubkey '{}' has wrong length: {} bytes (expected 32)",
                        s,
                        bytes.len()
                    );
                    None
                }
            },
            Err(e) => {
                error!("Failed to hex-decode trusted pubkey '{}': {}", s, e);
                None
            }
        })
        .collect()
}

/// Reads `TRUSTED_PUBKEYS` from the environment and parses it. An unset or
/// empty variable means verification is disabled — logged loudly since it's
/// a meaningful security posture change, not silently swallowed like the
/// original stub that just returned `vec![]` with no explanation.
fn load_trusted_keys() -> Vec<VerifyingKey> {
    match std::env::var("TRUSTED_PUBKEYS") {
        Ok(raw) if !raw.trim().is_empty() => {
            let keys = parse_trusted_keys(&raw);
            info!("Loaded {} trusted command-signing key(s)", keys.len());
            keys
        }
        _ => {
            warn!(
                "TRUSTED_PUBKEYS not set (or empty) — command signature verification is DISABLED"
            );
            vec![]
        }
    }
}

/// Verifies a command's contextual Ed25519 signature against the trusted
/// key set. Extracted from `handle_command` so it's testable without a
/// Kafka broker: given a command, the raw signature bytes from the
/// "x-signature" header, and the trusted key list, does *any* trusted key
/// validate the hash(robot_id ‖ command_id ‖ action ‖ sequence) digest?
///
/// An empty `trusted_keys` list means verification is disabled (matches
/// `load_trusted_keys`'s documented behavior when unset) and always passes
/// — this is a deliberate, logged opt-out, not a silent bypass.
fn verify_command_signature(
    cmd: &CommandEvent,
    sig_bytes: Option<&[u8]>,
    trusted_keys: &[VerifyingKey],
) -> Result<(), ProxyError> {
    if trusted_keys.is_empty() {
        return Ok(());
    }

    let sig_bytes = sig_bytes.ok_or(ProxyError::InvalidSignature)?;
    let sig = Signature::from_slice(sig_bytes).map_err(|_| ProxyError::InvalidSignature)?;

    let mut hasher = Sha256::new();
    hasher.update(cmd.robot_id.to_be_bytes());
    hasher.update(cmd.command_id);
    hasher.update(cmd.action.to_be_bytes());
    hasher.update(cmd.sequence.to_be_bytes());
    let message = hasher.finalize();

    let verified = trusted_keys
        .iter()
        .any(|key| key.verify(&message, &sig).is_ok());

    if verified {
        Ok(())
    } else {
        Err(ProxyError::InvalidSignature)
    }
}

type RobotRateLimiter =
    Arc<RateLimiter<u32, DefaultKeyedStateStore<u32>, governor::clock::DefaultClock>>;

#[tokio::main]
async fn main() -> Result<()> {
    tracing_subscriber::fmt().with_env_filter("info").init();

    let trusted_keys = load_trusted_keys();

    let consumer: StreamConsumer = rdkafka::ClientConfig::new()
        .set(
            "bootstrap.servers",
            std::env::var("KAFKA_BROKERS").unwrap_or("localhost:9092".into()),
        )
        .set("group.id", format!("robot-proxy-{}", ROBOT_ID))
        .set("enable.auto.commit", "false")
        .create()
        .context("Consumer creation failed")?;

    let producer: FutureProducer = rdkafka::ClientConfig::new()
        .set(
            "bootstrap.servers",
            std::env::var("KAFKA_BROKERS").unwrap_or("localhost:9092".into()),
        )
        .create()
        .context("Producer creation failed")?;

    consumer
        .subscribe(&[COMMANDS_TOPIC])
        .context("Subscribe failed")?;

    // Per-robot rate limiter: max 20 commands/sec
    let rate_limiter: RobotRateLimiter =
        Arc::new(RateLimiter::keyed(Quota::per_second(nonzero!(20u32))));

    let current_epoch: u64 = std::env::var("ROBOT_EPOCH")
        .ok()
        .and_then(|v| v.parse().ok())
        .unwrap_or(1);

    // V3-3: shared clock synchroniser (edge-proxy owns it exclusively)
    let mut clock_sync = ClockSync::new();

    // Real cross-process IPC rings: this proxy is the sole producer on
    // `ipc_rx_ring` (commands going to the microkernel) and the sole
    // consumer on `ipc_tx_ring` (telemetry coming from the microkernel).
    // See shm.rs for why `&'static SharedRingBuffer`, not `&mut`.
    let ipc_rx_ring_name = format!("/qib-ipc-rx-{}", ROBOT_ID);
    let ipc_tx_ring_name = format!("/qib-ipc-tx-{}", ROBOT_ID);
    let ipc_rx_ring: &'static SharedRingBuffer =
        unsafe { shm::map_shared_ring(&ipc_rx_ring_name)? };
    let ipc_tx_ring: &'static SharedRingBuffer =
        unsafe { shm::map_shared_ring(&ipc_tx_ring_name)? };

    // V3-2: Spawn event-driven telemetry exporter
    let producer_clone = producer.clone();
    tokio::spawn(async move {
        run_telemetry_exporter(producer_clone, ROBOT_ID, ipc_tx_ring).await;
    });

    let (shutdown_tx, mut shutdown_rx) = tokio::sync::watch::channel(false);
    tokio::spawn(async move {
        if let Ok(()) = signal::ctrl_c().await {
            info!("Shutdown signal received");
            let _ = shutdown_tx.send(true);
        }
    });

    info!("Edge proxy started for robot {}", ROBOT_ID);

    loop {
        tokio::select! {
            result = consumer.recv() => {
                match result {
                    Ok(m) => {
                        let result = handle_command(
                            &m,
                            &trusted_keys,
                            &rate_limiter,
                            current_epoch,
                            &mut clock_sync, // V3-3
                            ipc_rx_ring,
                        ).await;

                        if let Err(e) = result {
                            warn!("Command handling error: {}", e);
                            if let Some(payload) = m.payload() {
                                let _ = producer.send(
                                    FutureRecord::<(), [u8]>::to(qib_schema::DLQ_TOPIC).payload(payload),
                                    rdkafka::util::Timeout::Never,
                                ).await;
                            }
                        }

                        consumer.commit_message(&m, CommitMode::Async)
                            .unwrap_or_else(|e| error!("Commit failed: {}", e));
                    }
                    Err(e) => error!("Kafka recv error: {}", e),
                }
            }
            _ = shutdown_rx.changed() => {
                if *shutdown_rx.borrow() {
                    info!("Proxy shutting down cleanly");
                    break;
                }
            }
        }
    }

    // "Smaller fixes" round: `shm::unlink_shared_ring` existed but nothing
    // ever called it outside tests — a real process never cleaned up its
    // shm segments on exit. Whether that's actually desirable is a genuine
    // tradeoff, not an oversight to blindly "fix" by always unlinking:
    //
    //   - If the microkernel side is expected to keep running (or restart
    //     and reattach) independently of this proxy, unlinking on every
    //     proxy shutdown would mean a proxy restart always creates a fresh,
    //     empty ring instead of reattaching to one that may still hold
    //     in-flight state — silently discarding it.
    //   - For a single-owner-lifecycle deployment (proxy and microkernel
    //     side always restart together, or this is a dev/test environment),
    //     leaving stale named segments around after every shutdown is a
    //     real leak of `/dev/shm` entries over time.
    //
    // So this is opt-in via IPC_UNLINK_ON_SHUTDOWN, defaulting to "leave it
    // alone" (the safer default for a fleet where either side may restart
    // independently), rather than picking one behavior unconditionally.
    if unlink_on_shutdown_enabled() {
        info!("IPC_UNLINK_ON_SHUTDOWN set — removing this proxy's shm segments");
        shm::unlink_shared_ring(&ipc_rx_ring_name);
        shm::unlink_shared_ring(&ipc_tx_ring_name);
    }

    Ok(())
}

/// Pure, testable parse of the `IPC_UNLINK_ON_SHUTDOWN` opt-in flag —
/// see the shutdown-path comment above for why this defaults to `false`.
fn unlink_on_shutdown_enabled() -> bool {
    std::env::var("IPC_UNLINK_ON_SHUTDOWN")
        .map(|v| v.eq_ignore_ascii_case("true") || v == "1")
        .unwrap_or(false)
}

async fn handle_command(
    m: &rdkafka::message::BorrowedMessage<'_>,
    trusted_keys: &[VerifyingKey],
    rate_limiter: &RobotRateLimiter,
    current_epoch: u64,
    clock_sync: &mut ClockSync, // V3-3
    ipc_rx_ring: &SharedRingBuffer,
) -> Result<(), ProxyError> {
    let payload = m.payload().unwrap_or(&[]);

    // ── 1. V3-6: Contextual Ed25519 Signature Verification ──────────
    //
    // The signature covers hash(robot_id ‖ command_id ‖ action ‖ sequence).
    // This prevents cross-robot replay: a valid CoolDown for Robot A cannot
    // be presented to Robot B even if the raw bytes are identical.
    let cmd: CommandEvent = qib_schema::deserialize_msg(payload)?;
    let sig_bytes = extract_header(m, "x-signature");
    verify_command_signature(&cmd, sig_bytes.as_deref(), trusted_keys)?;

    // ── 2. V3-3: Clock-Adjusted TTL Check ────────────────────────────
    //
    // Update the drift model with this command's cloud-issued timestamp,
    // then use the drift-corrected "now" for the expiry comparison.
    let raw_now = raw_time_ns();
    clock_sync.synchronize(cmd.issued_at_ns, raw_now);
    let adjusted_now = clock_sync.adjusted_now_ns();

    if adjusted_now > cmd.valid_until_ns {
        return Err(ProxyError::ExpiredCommand {
            valid_until_ns: cmd.valid_until_ns,
            now_ns: adjusted_now,
        });
    }

    // ── 3. Epoch Check ────────────────────────────────────────────────
    if current_epoch < cmd.min_epoch || current_epoch > cmd.max_epoch {
        return Err(ProxyError::EpochMismatch {
            robot_epoch: current_epoch,
            min_epoch: cmd.min_epoch,
            max_epoch: cmd.max_epoch,
        });
    }

    // ── 4. Rate Limiting ──────────────────────────────────────────────
    rate_limiter
        .check_key(&cmd.robot_id)
        .map_err(|_| ProxyError::RateLimited {
            robot_id: cmd.robot_id,
        })?;

    // ── 5. Push to Microkernel IPC Ring (zero-copy) ──────────────────
    let pushed = ipc_rx_ring.push(IpcMessage::Command {
        action: cmd.action,
        duration_sec: cmd.duration_sec,
        valid_until_ns: cmd.valid_until_ns,
        issued_at_ns: cmd.issued_at_ns,
        min_epoch: cmd.min_epoch,
        max_epoch: cmd.max_epoch,
        sequence: cmd.sequence,
        command_id: cmd.command_id,
    });
    if !pushed {
        return Err(ProxyError::IpcOverflow);
    }

    info!("Command {:?} dispatched to microkernel", cmd.command_id);
    Ok(())
}

// ── V3-2: Event-Driven Telemetry Exporter ────────────────────────────
//
// Previous version: tokio::time::sleep(10ms) — worst-case 10ms latency.
// V3: yield_now() immediately hands control back to the async runtime
// and re-polls the ring on the very next scheduling opportunity.
// This reduces safety-response latency from ~10ms to ~microseconds.
async fn run_telemetry_exporter(
    producer: FutureProducer,
    robot_id: u32,
    ipc_tx_ring: &'static SharedRingBuffer,
) {
    loop {
        // 1. Try to pop from the IPC TX ring (wait-free)
        if let Some(msg) = poll_ipc_tx_ring(ipc_tx_ring, robot_id) {
            let payload = qib_schema::serialize_msg(&msg); // V3-1: postcard
            if let Err((e, _)) = producer
                .send(
                    FutureRecord::to(TELEMETRY_CONTROL_TOPIC)
                        .key(&robot_id.to_be_bytes())
                        .payload(&payload),
                    rdkafka::util::Timeout::After(std::time::Duration::from_millis(50)),
                )
                .await
            {
                error!("Telemetry send failed: {}", e);
            }
        } else {
            // 2. No data yet — yield to other tasks for one scheduler quantum,
            // then immediately re-check. Far faster than sleep(10ms).
            tokio::task::yield_now().await;
        }
    }
}

/// Polls the shared IPC TX ring buffer for a telemetry reading pushed by
/// the microkernel, translating the wire-level `IpcMessage::Telemetry`
/// (fixed-point, no timestamp/robot_id) into the full `TelemetryEvent` the
/// rest of the pipeline expects.
///
/// The microkernel signals an IPC-overflow condition (its own rx ring was
/// full when the proxy tried to push a command) via the sentinel z_velocity
/// value `-1.0` — see `qib_muk::scheduler::tick`'s telemetry-emit step.
fn poll_ipc_tx_ring(ipc_tx_ring: &SharedRingBuffer, robot_id: u32) -> Option<TelemetryEvent> {
    match ipc_tx_ring.pop()? {
        IpcMessage::Telemetry(temp_fixed, z_fixed) => {
            let overflow_sentinel = z_fixed == I20F44::from_num(-1.0);
            Some(TelemetryEvent {
                robot_id,
                timestamp_ns: raw_time_ns(),
                temp: temp_fixed.to_num::<f64>(),
                z_velocity: if overflow_sentinel {
                    0.0
                } else {
                    z_fixed.to_num::<f64>()
                },
                status: if overflow_sentinel { 3 } else { 0 },
                ack_command_id: None,
            })
        }
        // Empty / Command variants should never appear on the TX ring (the
        // microkernel only ever pushes Telemetry here); ignore defensively
        // rather than panicking on a malformed/foreign write.
        _ => None,
    }
}

fn extract_header(m: &rdkafka::message::BorrowedMessage<'_>, key: &str) -> Option<Vec<u8>> {
    m.headers()?.iter().find_map(|h| {
        if h.key == key {
            h.value.map(|v| v.to_vec())
        } else {
            None
        }
    })
}

fn raw_time_ns() -> u64 {
    SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
        .as_nanos() as u64
}

#[cfg(test)]
mod tests {
    use super::*;
    use ed25519_dalek::{Signer, SigningKey};
    use rand::rngs::OsRng;

    fn make_cmd(robot_id: u32, action: u8, sequence: u64) -> CommandEvent {
        CommandEvent {
            robot_id,
            command_id: [7u8; 16],
            sequence,
            action,
            duration_sec: 60,
            issued_at_ns: 0,
            valid_until_ns: u64::MAX,
            min_epoch: 1,
            max_epoch: u64::MAX,
        }
    }

    fn sign(cmd: &CommandEvent, key: &SigningKey) -> Signature {
        let mut hasher = Sha256::new();
        hasher.update(cmd.robot_id.to_be_bytes());
        hasher.update(cmd.command_id);
        hasher.update(cmd.action.to_be_bytes());
        hasher.update(cmd.sequence.to_be_bytes());
        key.sign(&hasher.finalize())
    }

    // ── parse_trusted_keys ────────────────────────────────────────────

    #[test]
    fn parse_trusted_keys_accepts_valid_hex_keys() {
        let key1 = SigningKey::generate(&mut OsRng).verifying_key();
        let key2 = SigningKey::generate(&mut OsRng).verifying_key();
        let raw = format!("{}, {}", hex::encode(key1.to_bytes()), hex::encode(key2.to_bytes()));

        let parsed = parse_trusted_keys(&raw);
        assert_eq!(parsed.len(), 2);
        assert_eq!(parsed[0].to_bytes(), key1.to_bytes());
        assert_eq!(parsed[1].to_bytes(), key2.to_bytes());
    }

    #[test]
    fn parse_trusted_keys_skips_malformed_entries_but_keeps_valid_ones() {
        let key = SigningKey::generate(&mut OsRng).verifying_key();
        let raw = format!(
            "not-hex-at-all,{},deadbeef",
            hex::encode(key.to_bytes())
        );
        let parsed = parse_trusted_keys(&raw);
        assert_eq!(parsed.len(), 1, "only the one valid 32-byte key should survive");
        assert_eq!(parsed[0].to_bytes(), key.to_bytes());
    }

    #[test]
    fn parse_trusted_keys_empty_string_yields_empty_list() {
        assert!(parse_trusted_keys("").is_empty());
        assert!(parse_trusted_keys("   ").is_empty());
    }

    // ── verify_command_signature ──────────────────────────────────────

    #[test]
    fn verification_disabled_when_trusted_keys_empty() {
        let cmd = make_cmd(1, 2, 3);
        // No signature bytes at all, and no trusted keys — must still pass,
        // matching load_trusted_keys' documented "unset = disabled" behavior.
        assert!(verify_command_signature(&cmd, None, &[]).is_ok());
    }

    #[test]
    fn valid_signature_from_a_trusted_key_verifies() {
        let signing_key = SigningKey::generate(&mut OsRng);
        let trusted = vec![signing_key.verifying_key()];
        let cmd = make_cmd(7, 1, 42);
        let sig = sign(&cmd, &signing_key);

        assert!(verify_command_signature(&cmd, Some(&sig.to_bytes()), &trusted).is_ok());
    }

    #[test]
    fn signature_for_a_different_robot_is_rejected() {
        // This is the core V3-6 property: a signature computed for robot A's
        // command must not verify when replayed against robot B's command,
        // even with the same trusted key and otherwise-identical fields.
        let signing_key = SigningKey::generate(&mut OsRng);
        let trusted = vec![signing_key.verifying_key()];

        let cmd_a = make_cmd(1, 2, 5);
        let sig_for_a = sign(&cmd_a, &signing_key);

        let cmd_b = make_cmd(2, 2, 5); // different robot_id, same action/sequence
        let result = verify_command_signature(&cmd_b, Some(&sig_for_a.to_bytes()), &trusted);
        assert!(matches!(result, Err(ProxyError::InvalidSignature)));
    }

    #[test]
    fn signature_from_an_untrusted_key_is_rejected() {
        let untrusted_key = SigningKey::generate(&mut OsRng);
        let trusted_key = SigningKey::generate(&mut OsRng);
        let trusted = vec![trusted_key.verifying_key()];

        let cmd = make_cmd(1, 1, 1);
        let sig = sign(&cmd, &untrusted_key);

        let result = verify_command_signature(&cmd, Some(&sig.to_bytes()), &trusted);
        assert!(matches!(result, Err(ProxyError::InvalidSignature)));
    }

    #[test]
    fn missing_signature_header_is_rejected_when_verification_enabled() {
        let signing_key = SigningKey::generate(&mut OsRng);
        let trusted = vec![signing_key.verifying_key()];
        let cmd = make_cmd(1, 1, 1);

        let result = verify_command_signature(&cmd, None, &trusted);
        assert!(matches!(result, Err(ProxyError::InvalidSignature)));
    }

    // ── ClockSync ("smaller fixes" round) ──────────────────────────────

    #[test]
    fn normal_drift_updates_the_ewma_without_clamping() {
        let mut cs = ClockSync::new();
        // Cloud thinks it's 500ns earlier than the proxy's receive time —
        // a perfectly ordinary, small amount of drift.
        cs.synchronize(1_000_000_000, 1_000_000_500);
        // (0 * 9 + (-500)) / 10 = -50
        assert_eq!(cs.drift_offset_ns, -50);
    }

    #[test]
    fn implausible_drift_is_clamped_to_the_bound() {
        let mut cs = ClockSync::new();
        // proxy_received_ns far in the future relative to cloud_issued_ns —
        // representative of a corrupted/malicious timestamp, not real skew.
        cs.synchronize(0, u64::MAX / 2);
        assert!(
            cs.drift_offset_ns.abs() <= MAX_PLAUSIBLE_DRIFT_NS,
            "drift_offset_ns ({}) exceeded the plausible bound ({})",
            cs.drift_offset_ns,
            MAX_PLAUSIBLE_DRIFT_NS
        );
    }

    #[test]
    fn repeated_pathological_input_never_overflows_or_panics() {
        let mut cs = ClockSync::new();
        // Alternate between the two most extreme u64 values across many
        // calls — if the old unbounded `drift_offset_ns * 9` were still in
        // place, this would eventually overflow i64 (panicking in a debug
        // build). The bound must hold on every single call, not just
        // eventually converge.
        for i in 0..10_000 {
            if i % 2 == 0 {
                cs.synchronize(u64::MAX, 0);
            } else {
                cs.synchronize(0, u64::MAX);
            }
            assert!(
                cs.drift_offset_ns.abs() <= MAX_PLAUSIBLE_DRIFT_NS,
                "drift_offset_ns escaped its bound at iteration {i}: {}",
                cs.drift_offset_ns
            );
        }
    }

    #[test]
    fn adjusted_now_ns_reflects_the_current_offset() {
        let mut cs = ClockSync::new();
        assert_eq!(cs.drift_offset_ns, 0);
        // With zero offset, adjusted_now_ns should be within a very tight
        // tolerance of raw_time_ns() (allowing for the few microseconds
        // that might elapse between the two calls in this test).
        let before = raw_time_ns();
        let adjusted = cs.adjusted_now_ns();
        let after = raw_time_ns();
        assert!(adjusted >= before && adjusted <= after);

        // A large-but-plausible negative offset shifts adjusted_now_ns
        // backward by roughly that amount.
        cs.drift_offset_ns = -1_000_000_000; // -1s
        let shifted = cs.adjusted_now_ns();
        assert!(shifted < raw_time_ns());
    }

    // ── unlink_on_shutdown_enabled ("smaller fixes" round) ─────────────

    /// `std::env::set_var` is process-global; guard against these three
    /// tests interleaving with each other under parallel test execution.
    static ENV_GUARD: std::sync::Mutex<()> = std::sync::Mutex::new(());

    #[test]
    fn unlink_on_shutdown_defaults_to_disabled() {
        let _guard = ENV_GUARD.lock().unwrap();
        std::env::remove_var("IPC_UNLINK_ON_SHUTDOWN");
        assert!(!unlink_on_shutdown_enabled());
    }

    #[test]
    fn unlink_on_shutdown_accepts_true_and_1() {
        let _guard = ENV_GUARD.lock().unwrap();
        for value in ["true", "TRUE", "True", "1"] {
            std::env::set_var("IPC_UNLINK_ON_SHUTDOWN", value);
            assert!(unlink_on_shutdown_enabled(), "expected '{value}' to enable it");
        }
        std::env::remove_var("IPC_UNLINK_ON_SHUTDOWN");
    }

    #[test]
    fn unlink_on_shutdown_rejects_other_values() {
        let _guard = ENV_GUARD.lock().unwrap();
        for value in ["false", "0", "yes", ""] {
            std::env::set_var("IPC_UNLINK_ON_SHUTDOWN", value);
            assert!(!unlink_on_shutdown_enabled(), "expected '{value}' to leave it disabled");
        }
        std::env::remove_var("IPC_UNLINK_ON_SHUTDOWN");
    }
